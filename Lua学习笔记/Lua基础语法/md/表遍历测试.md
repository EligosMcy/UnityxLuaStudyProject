## ipairs遍历

用于数组/序列（从1开始，连续索引）：

```lua
local fruits = {"苹果", "香蕉", "橙子", "葡萄"}
for i, v in ipairs(fruits) do
    print(i, v)
end
```

## pairs遍历

用于字典/键值对（无序）：

```lua
local person = {name = "张三", age = 25, city = "北京"}
for k, v in pairs(person) do
    print(k, v)
end
```

## 泛型for循环遍历

```lua
local nums = {10, 20, 30, 40, 50}
for i = 1, #nums do
    print("nums[" .. i .. "] =", nums[i])
end
```

## 遍历混合表

```lua
local mixed = {1, 2, 3, name = "李四", age = 30}

print("使用ipairs遍历混合表：")
for i, v in ipairs(mixed) do
    print(i, v)  -- 只输出1,2,3
end

print("使用pairs遍历混合表：")
for k, v in pairs(mixed) do
    print(k, v)  -- 输出所有键值对
end
```

## 遍历二维表

```lua
local matrix = {
    {1, 2, 3},
    {4, 5, 6},
    {7, 8, 9}
}

print("按行遍历：")
for i, row in ipairs(matrix) do
    local line = ""
    for j, val in ipairs(row) do
        line = line .. val .. " "
    end
    print("行" .. i .. ": " .. line)
end
```

## pairs遍历顺序说明

pairs遍历顺序是随机的，不是按照插入顺序：

```lua
local test = {}
for i = 1, 10 do
    test["key" .. i] = i
end

for k, v in pairs(test) do
    print(k, v)
end
```

## 嵌套表遍历

```lua
local students = {
    {name = "学生A", scores = {90, 85, 88}},
    {name = "学生B", scores = {75, 92, 80}},
    {name = "学生C", scores = {88, 76, 95}}
}

for i, student in ipairs(students) do
    print("学生: " .. student.name)
    for j, score in ipairs(student.scores) do
        print("  科目" .. j .. ": " .. score)
    end
end
```

## 遍历时修改表

不要在遍历时直接修改表，可能导致无限循环：

```lua
-- 错误做法：在遍历中添加元素
-- for i, v in ipairs(items) do
--     if v == 3 then
--         table.insert(items, 100)  -- 不要在遍历时修改表
--     end
-- end

-- 正确做法：遍历副本
local t_copy = {1, 2, 3, 4, 5}
for i, v in ipairs(t_copy) do
    if v == 3 then
        table.insert(t_copy, 100)
    end
end
```

## 遍历时删除元素

```lua
local items = {a = 1, b = 2, c = 3, d = 4}

local to_delete = {}
for k, v in pairs(items) do
    if v % 2 == 0 then
        table.insert(to_delete, k)
    end
end

for i, k in ipairs(to_delete) do
    items[k] = nil
end
```

## #操作符与ipairs的区别

```lua
local arr = {1, 2, 3}
print(#arr)  -- 3

local arr_with_nil = {1, 2, nil, 4, 5}
print(#arr_with_nil)  -- 2（遇到nil停止）

print("ipairs遍历（只遍历连续部分）:")
for i, v in ipairs(arr_with_nil) do
    print("i=" .. i .. ", v=" .. v)
end
```
