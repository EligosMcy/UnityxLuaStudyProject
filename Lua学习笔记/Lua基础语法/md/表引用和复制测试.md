## 引用类型演示

Lua中表是引用类型：

```lua
local t1 = {name = "原始表", value = 100}
local t2 = t1  -- t2只是t1的引用，不是副本

t2.name = "修改后的表"
print(t1.name)  -- 也变了！
print(t2.name)  -- 同样的变化

print(t1 == t2)  -- true，指向同一个表
```

## 函数参数传递

```lua
local function modifyTable(t)
    t.value = 999
    t.newField = "新字段"
end

local testTable = {value = 10}
modifyTable(testTable)
print(testTable.value)    -- 999
print(testTable.newField) -- 新字段
```

## 浅拷贝

```lua
function shallowCopy(original)
    local copy = {}
    for k, v in pairs(original) do
        copy[k] = v
    end
    return copy
end

local original = {name = "原始", nested = {a = 1, b = 2}}
local copied = shallowCopy(original)

copied.name = "副本"
print(original.name)  -- 不变

copied.nested.a = 100
print(original.nested.a)  -- 也变了！（问题所在）
```

## 深拷贝

```lua
function deepCopy(original)
    local copy = {}
    for k, v in pairs(original) do
        if type(v) == "table" then
            copy[k] = deepCopy(v)
        else
            copy[k] = v
        end
    end
    return copy
end

local original2 = {
    name = "原始",
    nested = {a = 1, b = 2},
    data = {inner = {value = 50}}
}
local deepCopied = deepCopy(original2)

deepCopied.nested.a = 999
print(original2.nested.a)  -- 不变！
```

## 循环引用的处理

```lua
local circular1 = {value = 1}
local circular2 = {value = 2}
circular1.next = circular2
circular2.next = circular1  -- 形成循环引用

function safeDeepCopy(original, visited)
    visited = visited or {}
    if type(original) ~= "table" then
        return original
    end

    if visited[original] then
        return visited[original]
    end

    local copy = {}
    visited[original] = copy

    for k, v in pairs(original) do
        copy[k] = safeDeepCopy(v, visited)
    end
    return copy
end

local circularCopy = safeDeepCopy(circular1)
print(circularCopy.next.value)
```

## 表作为函数返回值

```lua
function createCounter()
    local count = 0
    return {
        increment = function()
            count = count + 1
            return count
        end,
        getCount = function()
            return count
        end
    }
end

local counter1 = createCounter()
local counter2 = createCounter()

print(counter1.increment())  -- 1
print(counter1.increment())  -- 2
print(counter2.increment())  -- 1（独立的count）
```

## 表的比较

```lua
local a = {1, 2, 3}
local b = {1, 2, 3}
local c = a

print(a == b)  -- false（不同引用）
print(a == c)  -- true（同一引用）
```

## 表的清空

```lua
local toClear = {a = 1, b = 2, c = 3, d = 4}

-- 方法2：创建新表替换
toClear = {}  -- 原来的表失去引用，等待GC回收
print(toClear.a)  -- nil
```

## 多层引用

```lua
local level1 = {name = "第一层"}
local level2 = {parent = level1}
local level3 = {parent = level2}

print(level3.parent.parent.name)  -- 第一层

level1.name = "修改后的第一层"
print(level3.parent.parent.name)  -- 修改后的第一层
```
