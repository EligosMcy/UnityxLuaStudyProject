## 难点1：数组索引从1开始

Lua数组索引从1开始（不是0）：

```lua
local arr = {10, 20, 30, 40, 50}
print(arr[1])  -- 10（第一个元素）
print(arr[0])  -- nil
print(arr[2])  -- 20（第二个元素）
```

## 难点2：#操作符遇到nil就停止

```lua
local t1 = {1, 2, 3, 4, 5}
print(#t1)  -- 5

local t2 = {1, 2, nil, 4, 5}
print(#t2)  -- 2

-- 正确获取表元素个数的方法
local function getSize(t)
    local count = 0
    for _ in pairs(t) do
        count = count + 1
    end
    return count
end
print(getSize(t2))  -- 5
```

## 难点3：未赋值的键默认为nil

```lua
local dict = {name = "张三", age = 25}
print(dict.city)  -- nil

-- 判断键是否存在
if dict.name ~= nil then
    print("dict.name 存在")
end
```

## 难点4：ipairs vs pairs的区别

```lua
local mixed = {1, 2, 3, name = "李四", age = 30}

print("使用ipairs遍历（只遍历连续数组部分）:")
for i, v in ipairs(mixed) do
    print(i, v)  -- 只输出1,2,3
end

print("使用pairs遍历（遍历所有）:")
for k, v in pairs(mixed) do
    print(k, v)  -- 输出所有键值对
end
```

## 难点5：赋值是引用传递

```lua
local original = {value = 100}
local copy = original
copy.value = 200
print(original.value)  -- 200（被意外修改）

-- 正确拷贝
local correctCopy = {value = original.value}  -- 值拷贝
correctCopy.value = 300
print(original.value)  -- 200（不变）
```

## 难点6：遍历时修改表

```lua
-- 正确做法：先收集要添加的内容
local toAdd = {}
for i, v in ipairs(items) do
    if v == 2 then
        table.insert(toAdd, 6)
    end
end
for _, v in ipairs(toAdd) do
    table.insert(items, v)
end
```

## 难点7：元表查找链

```lua
local parent = {parentValue = "来自父类"}
local child = setmetatable({}, {__index = parent})

print(child.parentValue)  -- 来自父类

-- __index可以是一个函数
local obj = setmetatable({}, {
    __index = function(t, k)
        return "默认值:" .. k
    end
})
print(obj.anyKey)  -- 默认值:anyKey
```

## 难点8：表相等比较是引用比较

```lua
local a = {1, 2, 3}
local b = {1, 2, 3}
local c = a

print(a == b)  -- false（内容相同但不是同一引用）
print(a == c)  -- true（同一引用）
```

## 难点11：table库函数的副作用

table.sort会直接修改原表：

```lua
local t = {3, 1, 4, 1, 5, 9, 2, 6}
table.sort(t)  -- 直接修改原表

-- 解决：先拷贝再排序
local t2 = {3, 1, 4, 1, 5, 9, 2, 6}
local sorted = {}
for i, v in ipairs(t2) do
    sorted[i] = v
end
table.sort(sorted)
```

## 难点12：多返回值与表

```lua
function getValues()
    return 1, 2, 3

-- 将多返回值存入表
local packed = {getValues()}
print(table.concat(packed, ", "))  -- 1, 2, 3
```

## 难点13：使用表作为键

```lua
local TableAsKey = {}
setmetatable(TableAsKey, {
    __index = function(t, k)
        if type(k) == "table" then
            local key = tostring(k)
            return rawget(t, key)
        end
    end,
    __newindex = function(t, k, v)
        if type(k) == "table" then
            local key = tostring(k)
            rawset(t, key, v)
        end
    end
})

local key = {1, 2, 3}
TableAsKey[key] = "使用表作为键"
print(TableAsKey[key])
```

## 难点14：nil和空表的区别

```lua
local nilVar = nil
local emptyTable = {}

print(nilVar == nil)  -- true
print(#emptyTable)  -- 0
print(next(emptyTable))  -- nil（空表）

if next(emptyTable) == nil then
    print("空表")
end
```
