---1.1.1 点击宝箱领取金币钻石太拥挤，分散一点

local UnityEngine = CS.UnityEngine
local DefaultNamespace = CS.DefaultNamespace

xlua.hotfix(DefaultNamespace.HotfixLogic,'CreatePrize',function(self)
        for i = 0, 4 , 1 do

            --修改 30 -> 40
            local go = UnityEngine.GameObject.Instantiate(self.gold,self.transform.position + UnityEngine.Vecotr3(-10 + i * 30 , 0 , 0) , self.transform.rotation)
            go.transform.SetParent(go.transfomr , self.canvas)

            --修改 30 -> 40
            local go1 = UnityEngine.GameObject.Instantiate(self.diamand,self.transform.position + UnityEngine.Vecotr3(0 , 30 , 0) + UnityEngine.Vector3(-10 + i * 30 , 0 , 0) , self.transform.rotation)
            go1.transform.SetParent(go1.transfomr , self.canvas) 
        end

end)
