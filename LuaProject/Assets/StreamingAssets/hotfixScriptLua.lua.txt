local Ue = CS.UnityEngine
local UeInput = Ue.Input
local UeGameObj = Ue.GameObject
local HotFix = CS.HotFixScripts
local UeQuaternion = Ue.Quaternion

xlua.hotfix(HotFix.HotfixLogic, 'CreatePrize', function(self)
    for i = 0, 4, 1 do

        -- 修改 30 -> 40
        local go = UeGameObj.Instantiate(self.gold, self.transform.position + UnityEngine.Vecotr3(-10 + i * 30, 0, 0), self.transform.rotation)
        go.transform.SetParent(go.transfomr, self.canvas)

        -- 修改 30 -> 40
        local go1 = UeGameObj.Instantiate(self.diamand, self.transform.position + UnityEngine.Vecotr3(0, 30, 0) + UnityEngine.Vector3(-10 + i * 30, 0, 0), self.transform.rotation)
        go1.transform.SetParent(go1.transfomr, self.canvas)
    end

end )

            UeGameObj.Instantiate(self.Bullects[self.gunLevel - 1],self.attackPos.position,self.attackPos.rotation * UeQuaternion.Enler(0,0,20))
            UeGameObj.Instantiate(self.Bullects[self.gunLevel - 1],self.attackPos.position,self.attackPos.rotation * UeQuaternion.Enler(0,0,-20))

        UeGameObj.Instantiate(self.Bullects[self.gunLevel],self.attackPos.position,self.       ttackPos.rotation)

--util,这个文件是Xlua插件自带的需要将它放到你加载的文件夹中  [StreamingAssets]
local util = require 'util'

xlua.private_accessible(HotFix.HotFixLogic)

--使用 util 和 hotfix_ex,修改具体功能
util.hotfix_ex(HotFix.HotFixLogic,'Start',function(self)

    --这里注意,因为使用的C#中的方法所以需要 Start 添加 [LuaCallCSharp]
    self.Start(self)
    self.m_ReduceGold = self.m_ReduceGold - 20
end)


-- 1.4 当扣除数量不够后会出现负数
util.hotfix_ex(HotFix.HotFixLogic,'GoldChange',function(self number)
    self.GoldChange(self,number)
    if self.glod < -number then
        self.glod = 0
        return
    end
end)
